CREATE TABLE IF NOT EXISTS empresas (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID NOT NULL UNIQUE REFERENCES auth.users(id) ON DELETE CASCADE,
  nome_empresa TEXT NOT NULL,
  cnpj TEXT,
  regime TEXT DEFAULT 'mei',
  sn_anexo TEXT DEFAULT 'I',
  sn_rba DECIMAL(15,2) DEFAULT 180000,
  drive_url TEXT,
  criado_em TIMESTAMPTZ DEFAULT NOW()
);
ALTER TABLE empresas ENABLE ROW LEVEL SECURITY;
CREATE POLICY "usuario_ve_propria" ON empresas FOR ALL USING (auth.uid() = user_id);

CREATE TABLE IF NOT EXISTS notas_fiscais (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID NOT NULL REFERENCES auth.users(id) ON DELETE CASCADE,
  mes TEXT NOT NULL,
  ano INTEGER NOT NULL,
  nome_arquivo TEXT,
  total_itens INTEGER DEFAULT 0,
  total_custo DECIMAL(15,2) DEFAULT 0,
  dados JSONB,
  criado_em TIMESTAMPTZ DEFAULT NOW()
);
ALTER TABLE notas_fiscais ENABLE ROW LEVEL SECURITY;
CREATE POLICY "usuario_ve_propria_nf" ON notas_fiscais FOR ALL USING (auth.uid() = user_id);
