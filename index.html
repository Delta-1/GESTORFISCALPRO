Setup Supabase (1ª vez)

Execute o SQL abaixo no SQL Editor do seu projeto Supabase para criar as tabelas.

-- Empresas
create table if not exists empresas (
  id uuid primary key default gen_random_uuid(),
  user_id uuid references auth.users(id) not null,
  nome text, cnpj text, uf text, cidade text,
  regime text default 'SN',
  sn_anexo text default 'I',
  sn_rba numeric default 360000,
  created_at timestamptz default now()
);
alter table empresas enable row level security;
create policy "own" on empresas using (auth.uid()=user_id) with check (auth.uid()=user_id);

-- Notas fiscais
create table if not exists notas_fiscais (
  id uuid primary key default gen_random_uuid(),
  user_id uuid references auth.users(id) not null,
  empresa_id uuid references empresas(id),
  arquivo text, chave text,
  regime text, margem numeric, frete numeric,
  produtos jsonb,
  created_at timestamptz default now()
);
alter table notas_fiscais enable row level security;
create policy "own" on notas_fiscais using (auth.uid()=user_id) with check (auth.uid()=user_id);
